\title{Answers in a Matrix}

\begin{document}
\begin{abstract}
Answer boxes inside matrices and vectors.
\end{abstract}
\maketitle

Each entry is its own answer box with its own Check button. A screen reader
is told which entry a box is, for example ``answer, row 1, column 2''.

\begin{problem}
A matrix product:
\[
\begin{bmatrix} 1 & 2 \\ 3 & 4 \end{bmatrix}
\begin{bmatrix} 5 & 6 \\ 7 & 8 \end{bmatrix}
=
\begin{bmatrix} \answer{19} & \answer{22} \\ \answer{43} & \answer{50} \end{bmatrix}
\]
\end{problem}

\begin{problem}
A matrix times a vector:
\[
\begin{bmatrix} 2 & 0 \\ 1 & 3 \end{bmatrix}
\begin{bmatrix} 4 \\ 5 \end{bmatrix}
=
\begin{bmatrix} \answer{8} \\ \answer{19} \end{bmatrix}
\]
\end{problem}

\begin{problem}
A matrix with only some entries left blank. The transpose of
$\begin{bmatrix} 1 & 2 & 3 \\ 4 & 5 & 6 \\ 7 & 8 & 9 \end{bmatrix}$ is
\[
\begin{bmatrix} 1 & \answer{4} & 7 \\ \answer{2} & 5 & \answer{8} \\ 3 & 6 & 9 \end{bmatrix}
\]
\end{problem}

\begin{problem}
A wide one, to see how a matrix that does not fit is handled:
\[
I_4 =
\begin{bmatrix}
\answer{1} & \answer{0} & \answer{0} & \answer{0} \\
\answer{0} & \answer{1} & \answer{0} & \answer{0} \\
\answer{0} & \answer{0} & \answer{1} & \answer{0} \\
\answer{0} & \answer{0} & \answer{0} & \answer{1}
\end{bmatrix}
\]
\end{problem}
